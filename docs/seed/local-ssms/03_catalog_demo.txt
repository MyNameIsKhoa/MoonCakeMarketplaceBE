BEGIN TRY
    BEGIN TRANSACTION;

    INSERT INTO dbo.category (parent_category_id, name, slug, status)
    SELECT NULL, c.name, c.slug, N'ACTIVE' FROM @categories c
    WHERE NOT EXISTS (SELECT 1 FROM dbo.category existing WHERE existing.slug = c.slug);
    IF EXISTS (SELECT 1 FROM @categories c JOIN dbo.category existing ON existing.slug = c.slug
               WHERE existing.status <> N'ACTIVE' OR existing.parent_category_id IS NOT NULL)
        THROW 51103, 'A demo category is inactive; no existing category was changed.', 1;

    IF EXISTS (SELECT 1 FROM @stores d JOIN dbo.store s ON s.store_code = d.store_code
               WHERE s.supplier_id_ref <> d.supplier_id_ref
                  OR s.slug NOT IN (d.slug, SUBSTRING(d.slug, 6, 255)) OR s.status <> N'ACTIVE'
                  OR s.logo_url IS NULL OR s.banner_url IS NULL)
        THROW 51104, 'An existing demo store has different ownership, slug or status.', 1;
    IF EXISTS (SELECT 1 FROM @stores d JOIN dbo.store s ON s.slug = d.slug WHERE s.store_code <> d.store_code)
        THROW 51105, 'A demo store slug is already used by another store.', 1;
    INSERT INTO dbo.store (supplier_id_ref, store_code, name, slug, description, logo_url, banner_url, shipping_fee, status)
    SELECT d.supplier_id_ref, d.store_code, d.name, d.slug, d.description, d.logo_url, d.banner_url, d.shipping_fee, N'ACTIVE'
    FROM @stores d WHERE NOT EXISTS (SELECT 1 FROM dbo.store s WHERE s.store_code = d.store_code);

    IF EXISTS (SELECT 1 FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
               JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
               JOIN dbo.category c ON c.slug = d.category_slug
               WHERE p.category_id <> c.category_id OR p.status <> N'ACTIVE')
        THROW 51106, 'An existing demo product has a different category or status.', 1;
    INSERT INTO dbo.product (store_id, category_id, product_code, name, description, status)
    SELECT s.store_id, c.category_id, d.product_code, d.name, d.description, N'ACTIVE'
    FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
    JOIN dbo.category c ON c.slug = d.category_slug
    WHERE NOT EXISTS (SELECT 1 FROM dbo.product p WHERE p.store_id = s.store_id AND p.product_code = d.product_code);

    IF EXISTS (SELECT 1 FROM @products d JOIN dbo.product_variant v ON v.sku = d.sku
               JOIN dbo.product p ON p.product_id = v.product_id
               JOIN dbo.store s ON s.store_id = p.store_id
               WHERE p.product_code <> d.product_code OR s.store_code <> d.store_code OR v.status <> N'ACTIVE'
                  OR v.image_url IS NULL)
        THROW 51107, 'An existing demo SKU belongs to another product or is inactive.', 1;
    INSERT INTO dbo.product_variant (product_id, sku, variant_name, image_url, base_price, stock_quantity, status)
    SELECT p.product_id, d.sku, d.variant_name, d.image_url, d.base_price, d.stock_quantity, N'ACTIVE'
    FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
    JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
    WHERE NOT EXISTS (SELECT 1 FROM dbo.product_variant v WHERE v.sku = d.sku);

    UPDATE s SET name = d.name, description = d.description,
        shipping_fee = COALESCE(s.shipping_fee, d.shipping_fee)
    FROM dbo.store s JOIN @stores d ON d.store_code = s.store_code;
    UPDATE p SET name = d.name, description = d.description
    FROM dbo.product p JOIN dbo.store s ON s.store_id = p.store_id
    JOIN @products d ON d.store_code = s.store_code AND d.product_code = p.product_code;
    UPDATE v SET variant_name = d.variant_name
    FROM dbo.product_variant v JOIN @products d ON d.sku = v.sku;

    IF (SELECT COUNT(*) FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
        JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
        JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = d.sku) <> 10
        THROW 51108, 'Demo catalog verification failed.', 1;

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH;

SELECT c.slug AS category_slug, COUNT(p.product_id) AS demo_product_count
FROM @products d JOIN dbo.category c ON c.slug = d.category_slug
JOIN dbo.store s ON s.store_code = d.store_code
JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
GROUP BY c.slug ORDER BY c.slug;
GO

               WHERE c.category_id IS NULL OR c.status <> N'ACTIVE')
        THROW 51303, 'Run the base demo category seed first.', 1;
    IF EXISTS (SELECT 1 FROM @stores d JOIN dbo.store s ON s.store_code = d.store_code
               WHERE s.supplier_id_ref <> d.supplier_id_ref
                  OR s.slug NOT IN (d.slug, SUBSTRING(d.slug, 6, 255)) OR s.status <> N'ACTIVE')
        THROW 51304, 'An existing demo brand store conflicts with the seed.', 1;
    IF EXISTS (SELECT 1 FROM @stores d JOIN dbo.store s ON s.slug = d.slug WHERE s.store_code <> d.store_code)
        THROW 51305, 'A demo brand slug belongs to another store.', 1;

    UPDATE s SET name = d.name, description = d.description, logo_url = d.logo_url, banner_url = d.banner_url,
        shipping_fee = COALESCE(s.shipping_fee, d.shipping_fee)
    FROM dbo.store s JOIN @stores d ON d.store_code = s.store_code;

    INSERT INTO dbo.store (supplier_id_ref, store_code, name, slug, description, logo_url, banner_url, shipping_fee, status)
    SELECT d.supplier_id_ref, d.store_code, d.name, d.slug, d.description, d.logo_url, d.banner_url, d.shipping_fee, N'ACTIVE'
    FROM @stores d WHERE NOT EXISTS (SELECT 1 FROM dbo.store s WHERE s.store_code = d.store_code);

    IF EXISTS (SELECT 1 FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
               JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
               JOIN dbo.category c ON c.slug = d.category_slug
               WHERE p.category_id <> c.category_id OR p.status <> N'ACTIVE')
        THROW 51306, 'An existing demo brand product conflicts with the seed.', 1;
    INSERT INTO dbo.product (store_id, category_id, product_code, name, description, status)
    SELECT s.store_id, c.category_id, d.product_code, d.name, d.description, N'ACTIVE'
    FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
    JOIN dbo.category c ON c.slug = d.category_slug
    WHERE NOT EXISTS (SELECT 1 FROM dbo.product p WHERE p.store_id = s.store_id AND p.product_code = d.product_code);

    IF EXISTS (SELECT 1 FROM @products d JOIN dbo.product_variant v ON v.sku = d.sku
               JOIN dbo.product p ON p.product_id = v.product_id JOIN dbo.store s ON s.store_id = p.store_id
               WHERE p.product_code <> d.product_code OR s.store_code <> d.store_code OR v.status <> N'ACTIVE')
        THROW 51307, 'An existing demo brand SKU conflicts with the seed.', 1;
    INSERT INTO dbo.product_variant (product_id, sku, variant_name, image_url, base_price, stock_quantity, status)
    SELECT p.product_id, d.sku, d.variant_name, d.image_url, d.base_price, d.stock_quantity, N'ACTIVE'
    FROM @products d JOIN dbo.store s ON s.store_code = d.store_code
    JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
    WHERE NOT EXISTS (SELECT 1 FROM dbo.product_variant v WHERE v.sku = d.sku);

    UPDATE s SET name = d.name, description = d.description FROM dbo.store s JOIN @stores d ON d.store_code = s.store_code;
    UPDATE p SET name = d.name, description = d.description
    FROM dbo.product p JOIN dbo.store s ON s.store_id = p.store_id
    JOIN @products d ON d.store_code = s.store_code AND d.product_code = p.product_code;
    UPDATE v SET variant_name = d.variant_name
    FROM dbo.product_variant v JOIN @products d ON d.sku = v.sku;

        JOIN dbo.product p ON p.store_id = s.store_id AND p.product_code = d.product_code
        JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = d.sku) <> 24

    FROM @promotions d JOIN dbo.store s ON s.store_code = d.store_code
    WHERE NOT EXISTS (SELECT 1 FROM dbo.promotion pr WHERE pr.store_id = s.store_id AND pr.name = d.promo_name);

    INSERT INTO dbo.promotion_rule (promotion_id, target_variant_id, reward_variant_id, rule_type, min_quantity, discount_percent, discount_amount, reward_quantity)
    SELECT pr.promotion_id, NULL, NULL, d.rule_type, d.min_quantity, d.discount_percent, d.discount_amount, NULL
    FROM @promotions d
    JOIN dbo.store s ON s.store_code = d.store_code
    JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = d.promo_name
    WHERE NOT EXISTS (SELECT 1 FROM dbo.promotion_rule r WHERE r.promotion_id = pr.promotion_id);

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH;
GO
