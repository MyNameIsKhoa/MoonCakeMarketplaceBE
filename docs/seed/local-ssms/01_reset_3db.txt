-- CHI CHAY TREN SQL SERVER LOCAL DEMO. XOA VĨNH VIỄN dữ liệu của đúng 3 DB bên dưới.
-- Dừng Identity/Catalog/Commerce/Gateway trong IntelliJ trước khi chạy.
USE master;
GO
IF DB_ID(N'mss_commerce') IS NOT NULL
BEGIN
    ALTER DATABASE mss_commerce SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE mss_commerce;
END;
IF DB_ID(N'mss_catalog_store') IS NOT NULL
BEGIN
    ALTER DATABASE mss_catalog_store SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE mss_catalog_store;
END;
IF DB_ID(N'mss_identity') IS NOT NULL
BEGIN
    ALTER DATABASE mss_identity SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE mss_identity;
END;
GO
CREATE DATABASE mss_identity;
GO
CREATE DATABASE mss_catalog_store;
GO
CREATE DATABASE mss_commerce;
GO
-- Sau đây CHẠY BE: Identity -> Catalog -> Commerce để Flyway tạo bảng.
-- Khi cả 3 app đã Started, chạy 02_identity_demo.txt rồi 03_catalog_demo.txt.
SELECT name FROM sys.databases WHERE name IN (N'mss_identity',N'mss_catalog_store',N'mss_commerce');
