);

-- A partial seed must fail before changing anything. Every SKU must belong to its intended store.
IF (SELECT COUNT(*) FROM @offers o
    JOIN dbo.store s ON s.store_code = o.store_code AND s.status = N'ACTIVE'
    JOIN dbo.product p ON p.store_id = s.store_id AND p.status = N'ACTIVE'
    JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = o.sku AND v.status = N'ACTIVE'
    WHERE v.base_price > 0) <> (SELECT COUNT(*) FROM @offers)
    THROW 51502, 'Missing/inactive seeded store or SKU; run the earlier seeds or inspect the changed product.', 1;

IF EXISTS (SELECT 1 FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
    JOIN dbo.store other ON other.slug = SUBSTRING(s.slug, 6, 255) AND other.store_id <> s.store_id
    WHERE s.slug LIKE N'demo-%')
    THROW 51503, 'A clean store slug is already used by another store.', 1;

IF EXISTS (SELECT 1 FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
    JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = o.name
    GROUP BY s.store_id, o.name HAVING COUNT(*) > 1)
    THROW 51504, 'Duplicate named promotion; resolve it before re-running this seed.', 1;

BEGIN TRY
    BEGIN TRANSACTION;

    UPDATE s SET slug = SUBSTRING(s.slug, 6, 255), name = REPLACE(s.name, N' (demo)', N'')
    FROM dbo.store s JOIN @offers o ON o.store_code = s.store_code WHERE s.slug LIKE N'demo-%';
    UPDATE s SET name = REPLACE(s.name, N' (demo)', N'')
    FROM dbo.store s JOIN @offers o ON o.store_code = s.store_code WHERE s.name LIKE N'% (demo)';
    UPDATE s SET description = REPLACE(s.description, N' (demo)', N'')
    FROM dbo.store s JOIN @offers o ON o.store_code = s.store_code
    WHERE s.description LIKE N'% (demo)%';
    UPDATE p SET name = REPLACE(p.name, N' (demo)', N'')
    FROM dbo.product p JOIN dbo.store s ON s.store_id = p.store_id
    JOIN @offers o ON o.store_code = s.store_code WHERE p.name LIKE N'% (demo)';
    UPDATE p SET description = REPLACE(p.description, N' (demo)', N'')
    FROM dbo.product p JOIN dbo.store s ON s.store_id = p.store_id
    JOIN @offers o ON o.store_code = s.store_code WHERE p.description LIKE N'% (demo)%';
    UPDATE v SET variant_name = REPLACE(v.variant_name, N' (demo)', N'')
    FROM dbo.product_variant v JOIN dbo.product p ON p.product_id = v.product_id
    JOIN dbo.store s ON s.store_id = p.store_id
    JOIN @offers o ON o.store_code = s.store_code WHERE v.variant_name LIKE N'% (demo)';

    -- Retire only the old store-wide seed rule; do not touch seller-created promotions.
    UPDATE pr SET status = N'DISABLED'
    FROM dbo.promotion pr JOIN dbo.store s ON s.store_id = pr.store_id
    JOIN @offers o ON o.store_code = s.store_code AND o.old_name = pr.name
    WHERE EXISTS (SELECT 1 FROM dbo.promotion_rule r
                  WHERE r.promotion_id = pr.promotion_id AND r.target_variant_id IS NULL);

    -- Retire the previous product discounts only when the rule matches the old seed.
    UPDATE pr SET status = N'DISABLED'
    FROM dbo.promotion pr JOIN dbo.store s ON s.store_id = pr.store_id
    JOIN @offers o ON o.store_code = s.store_code AND o.previous_name = pr.name
    JOIN dbo.product_variant v ON v.sku = o.sku
    WHERE (SELECT COUNT(*) FROM dbo.promotion_rule r WHERE r.promotion_id = pr.promotion_id) = 1
      AND EXISTS (SELECT 1 FROM dbo.promotion_rule r WHERE r.promotion_id = pr.promotion_id
          AND r.rule_type = N'PERCENT_DISCOUNT' AND r.target_variant_id = v.variant_id AND r.min_quantity = 1);

    DECLARE @starts DATETIME2(0) = DATEADD(DAY, -1, SYSUTCDATETIME());
    DECLARE @ends DATETIME2(0) = DATEADD(DAY, 30, SYSUTCDATETIME());
    UPDATE pr SET start_at = @starts, end_at = @ends, status = N'ACTIVE'
    FROM dbo.promotion pr JOIN dbo.store s ON s.store_id = pr.store_id
    JOIN @offers o ON o.store_code = s.store_code AND o.name = pr.name;
    INSERT INTO dbo.promotion (store_id, name, start_at, end_at, status)
    SELECT s.store_id, o.name, @starts, @ends, N'ACTIVE'
    FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
    WHERE NOT EXISTS (SELECT 1 FROM dbo.promotion pr WHERE pr.store_id = s.store_id AND pr.name = o.name);

    IF EXISTS (SELECT 1 FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
        JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = o.name
        JOIN dbo.promotion_rule r ON r.promotion_id = pr.promotion_id
        GROUP BY pr.promotion_id HAVING COUNT(*) > 1)
        THROW 51505, 'A seeded product promotion has multiple rules; no rule was overwritten.', 1;

    UPDATE r SET target_variant_id = v.variant_id, reward_variant_id = v.variant_id,
        rule_type = N'BUY_X_GET_Y', min_quantity = o.buy_quantity,
        discount_percent = NULL, discount_amount = NULL, reward_quantity = o.gift_quantity
    FROM dbo.promotion_rule r JOIN dbo.promotion pr ON pr.promotion_id = r.promotion_id
    JOIN dbo.store s ON s.store_id = pr.store_id
    JOIN @offers o ON o.store_code = s.store_code AND o.name = pr.name
    JOIN dbo.product p ON p.store_id = s.store_id
    JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = o.sku;

    INSERT INTO dbo.promotion_rule (promotion_id, target_variant_id, reward_variant_id,
                                    rule_type, min_quantity, discount_percent, discount_amount, reward_quantity)
    SELECT pr.promotion_id, v.variant_id, v.variant_id, N'BUY_X_GET_Y', o.buy_quantity, NULL, NULL, o.gift_quantity
    FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
    JOIN dbo.product p ON p.store_id = s.store_id
    JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = o.sku
    JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = o.name
    WHERE NOT EXISTS (SELECT 1 FROM dbo.promotion_rule r WHERE r.promotion_id = pr.promotion_id);

    IF (SELECT COUNT(*) FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
        JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = o.name AND pr.status = N'ACTIVE'
        JOIN dbo.promotion_rule r ON r.promotion_id = pr.promotion_id AND r.min_quantity = o.buy_quantity
        JOIN dbo.product_variant v ON v.variant_id = r.target_variant_id AND v.sku = o.sku
        WHERE r.rule_type = N'BUY_X_GET_Y' AND r.reward_variant_id = v.variant_id
          AND r.reward_quantity = o.gift_quantity)
        <> (SELECT COUNT(*) FROM @offers)
        THROW 51506, 'Promotion verification failed; transaction rolled back.', 1;

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH;

SELECT s.name AS shop, s.slug, p.name AS product, v.sku, pr.name AS promotion,
       r.min_quantity AS buy_quantity, r.reward_quantity AS gift_quantity, pr.start_at, pr.end_at
FROM @offers o JOIN dbo.store s ON s.store_code = o.store_code
JOIN dbo.product p ON p.store_id = s.store_id
JOIN dbo.product_variant v ON v.product_id = p.product_id AND v.sku = o.sku
JOIN dbo.promotion pr ON pr.store_id = s.store_id AND pr.name = o.name
JOIN dbo.promotion_rule r ON r.promotion_id = pr.promotion_id AND r.target_variant_id = v.variant_id
ORDER BY s.name;
